%% ELSEVIER ADAPTER — HOST RESERVATION SYNC AFTER THE BAND'S TOP AREA
%% The adapter's \twocolumn[...] resets \@colroom to \@colht. NeoPage's contract
%% for column-mode switches is to call \spacereserve@syncaftermodeswitch next, so
%% reserved float space is re-applied; without it, AA_9001 overflowed 454pt.
%% The stand-in records that the hook ran on the band page, after the top area.
%% ASSERTION: the host sync runs once, on page 2, after the band's top area;
%% ASSERTION: HOST-SYNC-BAND begins page 2; paras PRE-01..POST-02 once; 2 pages.
%% =====================================================================
\documentclass[twocolumn]{article}
\makeatletter
\newcount\spacereserve@widegatetransaction
\newcount\fixture@syncs
\newcommand\spacereserve@syncaftermodeswitch{%
  \global\advance\fixture@syncs\@ne
  \ifnum\c@page=\tw@\else
    \PackageError{fixture}{host sync ran on page \the\c@page, not the band page}{}\fi
  \ifx\@dbltoplist\@empty
    \PackageError{fixture}{host sync ran before the band's top area existed}{}\fi}
\makeatother

\pagestyle{empty}
\setlength\parindent{0pt}
\newcommand\Body{Lorem ipsum dolor sit amet, consectetur adipiscing elit.}

\end{wideband}
\makeatletter
\ifnum\fixture@syncs=\@ne\else
