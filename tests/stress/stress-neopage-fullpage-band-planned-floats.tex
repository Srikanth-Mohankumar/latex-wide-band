%% NEOPAGE HOST — A PAGE OF ITS OWN FOR A BAND TAKES A PLANNED FLOAT PAGE (operator error)
%% NeoPage plans floats by physical page (satellite.json, floatReorder). A full-page band
%% ships a page the plan does not know, and the floats planned for that page number were
%% dropped with no error (NeoPage KNOSYS_112150: Figs 1-2; YJEMA_126557: Fig. 2, Table 2).
%% The operator now gets an error naming those floats. Stand-in: a loaded plan (the
%% fields neopage-float-placement.lua sets) putting fig1 on page 2 and tbl1 on page 9.
%% Band A ([fullpage]) is set on page 2 -> error naming fig1. Band B ([fullpage]) is set
%% on a page with nothing planned -> no error.
%% EXPECT-ERROR: band 1 (input line 34) is set on page 2, a page of its own, but NeoPage's float plan puts fig1 on page 2
%% ASSERTION: FP-BAND-A alone on page 2, FP-BAND-B alone on a later page; no other TeX
%% error; paras FP-01..FP-12 once.
%% =====================================================================
\begin{filecontents*}[overwrite]{\jobname-plan.json}
{"floatReorder": {"floatPlacementOrder": [
  {"page": 2, "column": "column1", "position": "top", "floatMeta": {"no": "fig1"}},
  {"page": 9, "column": "column1", "position": "top", "floatMeta": {"no": "tbl1"}},
]}}
\end{filecontents*}
